package "MiniDocker.Cli" {
  class Program {
    + Main(args: string[]) : int
  }
}

package "MiniDocker.Nucleo.Modelos" {
  enum EstadoConteiner {
    Criado
    Executando
    Parado
    Finalizado
  }

  class Conteiner {
    + Id : string
    + Nome : string
    + Comando : string
    + Estado : EstadoConteiner
    + CriadoEm : DateTime
    + IniciadoEm : DateTime?
    + FinalizadoEm : DateTime?
    + CaminhoRootFs : string
    + IdProcesso : int?
    + CodigoSaida : int?
    + Limites : LimitesRecursos
    + MemoriaUsadaBytes : long?
    + TempoCpuMs : double?
    + TotalProcessos : int?
    + FalhasDePagina : long?
  }

  class LimitesRecursos <<record>> {
    + MemoriaMaximaBytes : long?
    + ProcessosMaximos : int?
    + PercentualMaximoCpu : int?
    + {static} Nenhum : LimitesRecursos
    + AlgumDefinido : bool
    + Validar() : void
  }

  class ContabilidadeTrabalho <<record>> {
    + TempoCpuMs : double
    + PicoMemoriaBytes : long
    + TotalProcessos : int
    + FalhasDePagina : long
  }

  Conteiner "1" -- "1" EstadoConteiner
  Conteiner "1" *-- "1" LimitesRecursos
}

package "MiniDocker.Nucleo.Repositorios" {
  interface IRepositorioConteiner {
    + ObterTodos() : IReadOnlyList<Conteiner>
    + ObterPorNome(nome: string) : Conteiner?
    + Salvar(conteiner: Conteiner) : void
    + Remover(nome: string) : void
  }

  class RepositorioConteinerArquivo {
    - caminhoArquivo : string
    - sincronizador : object
    + ObterTodos() : IReadOnlyList<Conteiner>
    + ObterPorNome(nome: string) : Conteiner?
    + Salvar(conteiner: Conteiner) : void
    + Remover(nome: string) : void
  }

  RepositorioConteinerArquivo ..|> IRepositorioConteiner
  RepositorioConteinerArquivo ..> Conteiner : persiste em JSON
}

package "MiniDocker.Nucleo.Interop" {
  class ObjetoTrabalho {
    - handle : SafeJobHandle
    + ObjetoTrabalho(limites: LimitesRecursos)
    + Atribuir(processo: Process) : bool
    + Contabilizar() : ContabilidadeTrabalho
    + Encerrar(codigoSaida: int) : void
    + Dispose() : void
    - AplicarLimitesEstendidos(limites: LimitesRecursos) : void
    - AplicarLimiteCpu(limites: LimitesRecursos) : void
    - AplicarRestricoesUI() : void
  }

  class SafeJobHandle {
    # ReleaseHandle() : bool
  }

  class NativeMethods <<static>> {
    CreateJobObjectW()
    AssignProcessToJobObject()
    SetInformationJobObject()
    QueryInformationJobObject()
    TerminateJobObject()
    CloseHandle()
  }

  ObjetoTrabalho *-- SafeJobHandle
  ObjetoTrabalho ..> NativeMethods : P/Invoke
  ObjetoTrabalho ..> LimitesRecursos : aplica
  ObjetoTrabalho ..> ContabilidadeTrabalho : produz
}

package "MiniDocker.Nucleo.Servicos" {
  interface IServicoConteiner {
    + Executar(nome: string, comando: string, limites: LimitesRecursos?) : Conteiner
    + Listar() : IReadOnlyList<Conteiner>
    + Parar(nome: string) : void
    + Remover(nome: string) : void
  }

  class ServicoConteiner {
    - repositorio : IRepositorioConteiner
    - raizWorkspace : string
    + Executar(nome: string, comando: string, limites: LimitesRecursos?) : Conteiner
    + Listar() : IReadOnlyList<Conteiner>
    + Parar(nome: string) : void
    + Remover(nome: string) : void
    - CriarProcesso(comando: string, caminhoRootFs: string) : Process
    - IsolarAmbiente(informacoes: ProcessStartInfo, caminhoRootFs: string) : void
  }

  ServicoConteiner ..|> IServicoConteiner
  ServicoConteiner --> IRepositorioConteiner : usa
  ServicoConteiner ..> Conteiner : cria/gerencia
  ServicoConteiner ..> ObjetoTrabalho : cria um por execução
}

Program --> IServicoConteiner : usa

note right of ObjetoTrabalho
  Envólucro do Job Object do Windows.
  O kernel impõe as restrições de UI
  e o KILL_ON_JOB_CLOSE; a contabilidade
  cobre o grupo inteiro e sobrevive ao
  término dos processos.
end note

note right of ServicoConteiner
  Executa cmd.exe /c dentro do Job Object,
  com ambiente mínimo (sem herdar o host)
  e WorkingDirectory = rootfs.
  Sem isolamento de PID, rede ou
  sistema de arquivos.
end note
